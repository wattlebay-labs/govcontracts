<!-- Boot. A module rather than a classic inline script, because an inline classic script cannot be
     deferred: as a module it runs after the deferred scripts above, so Blazor is defined. Module scope keeps its helpers out of window; window.app is the API. -->
<script type="module">
    let queue = [];
    let activeRequests = 0;
    const maxConcurrent = 50;
    const maxRetries = 5;
    const retryDelay = 2000; // 2 seconds

    // Read and parse the importmap (safe fallback to empty object)
    const importMapEl = document.querySelector('script[type="importmap"]');
    const assetMap = (function () {
        try {
            if (!importMapEl || !importMapEl.textContent) return {};
            const parsed = JSON.parse(importMapEl.textContent);
            return (parsed && parsed.imports) ? parsed.imports : {};
        } catch (err) {
            console.error('Failed to parse importmap:', err);
            return {};
        }
    })();

    window.app = window.app || {};
    /**
     * Resolve an asset name against the parsed import map `assetMap`.
     *
     * Behavior:
     * 1. Try the name exactly as-provided.
     * 2. Normalize by stripping leading `./` or `/` and try three variants:
     *    - `./{base}`
     *    - `/{base}`
     *    - `{base}`
     * 3. If a match is found in `assetMap` return the mapped value.
     * 4. Otherwise return the original `name`.
     *
     * @param {string} name - The requested asset identifier (e.g. '/js/gtm.js' or './js/gtm.js').
     * @returns {string} Resolved asset URL from `assetMap`, or the original `name` if no mapping exists.
     */
    window.app.getAsset = (name) => {
        const asIs = String(name).trim();
        if (assetMap[asIs]) return assetMap[asIs];

        const stripLeading = (s) => s.replace(/^(\.\/|\/)+/, '');
        const base = stripLeading(asIs);

        const candidates = [
            './' + base,
            '/' + base,
            base
        ];

        for (const c of candidates) {
            if (assetMap[c]) return assetMap[c];
        }

        return name;
    };
    
    /**
     * Resolves once every data-before-reveal stylesheet in <head> has applied, or failed. Each is a
     * preload whose onload switches it to rel="stylesheet", which fires load a second time once
     * the sheet applies; onerror marks it failed. Those attributes record the state on the element
     * because this script runs after parsing, by when either event may already have fired.
     */
    const stylesApplied = Promise.all([...document.querySelectorAll('link[data-before-reveal]')].map(link =>
        new Promise(resolve => {
            const settled = () => (link.rel === 'stylesheet' && link.sheet) || link.dataset.beforeReveal === 'failed';
            if (settled()) return resolve();
            link.addEventListener('load', () => settled() && resolve());
            link.addEventListener('error', resolve);
        })));

    /**
     * Dismiss the initial loader. Called by MainLayout once the layout has rendered with its
     * real content - NOT when Blazor.start() resolves. Blazor.start() resolves as soon as the
     * runtime is up and root components have had their first render, at which point MainLayout,
     * the routed page and the header are all still awaiting HTTP, so revealing there uncovered
     * a half-built page that then visibly filled in.
     *
     * Idempotent: the failsafe timer, the boot-error path and the app itself all race to call it.
     */
    let revealed = false;
    window.app.reveal = () => {
        if (revealed) return;
        revealed = true;
        // Wait for the data-before-reveal stylesheets, but never strand the loader on one that
        // stalls. They are normally applied long before the runtime has even booted.
        const cap = new Promise(resolve => setTimeout(resolve, 3000));
        Promise.race([stylesApplied, cap]).then(() => {
            document.documentElement.classList.add('blazor-ready');
            // Outlast the 250ms opacity transition in app.css before removing the node.
            setTimeout(() => document.getElementById('initial-loader')?.remove(), 400);
            // Off the first visit's critical path: wait until the page has settled and the browser
            // is idle. Prefetch downloads at the lowest priority and does not execute anything.
            if (!window.app.isTelerikRoute) {
                const whenIdle = window.requestIdleCallback ?? (cb => setTimeout(cb, 0));
                setTimeout(() => whenIdle(window.app.prefetchTelerik), 5000);
            }
        });
    };

    Blazor.start({
        loadBootResource: function (type, name, defaultUri, integrity) {
            // Only batch wasm files (your 200 assemblies)
            if (!name.endsWith('.wasm')) {
                return defaultUri;
            }

            // Batch the wasm files
            return new Promise((resolve, reject) => {
                queue.push({
                    defaultUri,
                    integrity,
                    resolve,
                    reject,
                    retryCount: 0
                });
                processQueue();
            });
        }
    }).then(() => { // NOSONAR
        // Failsafe only: if a content fetch stalls or throws, the app may never call reveal()
        // and the loader would sit there forever. Reveal an imperfect page rather than nothing.
        setTimeout(window.app.reveal, 5000);
    }).catch((err) => {
        // A boot failure previously left the loader up permanently over a hidden #app,
        // i.e. a blank white page with no error surfaced.
        console.error('Blazor failed to start:', err);
        window.app.reveal();
    });

    function processQueue() {
        while (queue.length > 0 && activeRequests < maxConcurrent) {
            const item = queue.shift();
            activeRequests++;

            fetch(item.defaultUri, {
                cache: 'default',
                integrity: item.integrity
            })
                .then(response => {
                    if (response.status === 504) {
                        // Rate limit hit - retry after delay
                        if (item.retryCount < maxRetries) {
                            item.retryCount++;
                            console.log(`504 error for ${item.defaultUri}, retry ${item.retryCount}/${maxRetries} in ${retryDelay}ms`);

                            setTimeout(() => {
                                queue.push(item); // Re-queue at the end
                                processQueue();
                            }, retryDelay);

                            activeRequests--;
                            return;
                        } else {
                            throw new Error(`Failed to load ${item.defaultUri} after ${maxRetries} retries: 504 Gateway Timeout`);
                        }
                    }

                    if (!response.ok) {
                        throw new Error(`Failed to load ${item.defaultUri}: ${response.status}`);
                    }

                    item.resolve(response);
                    activeRequests--;
                    processQueue();
                })
                .catch(err => {
                    // Network errors might also benefit from retry
                    if (item.retryCount < maxRetries && (err.message.includes('504') || err.message.includes('Failed to fetch'))) {
                        item.retryCount++;
                        console.log(`Network error for ${item.defaultUri}, retry ${item.retryCount}/${maxRetries} in ${retryDelay}ms`);

                        setTimeout(() => {
                            queue.push(item);
                            processQueue();
                        }, retryDelay);

                        activeRequests--;
                    } else {
                        item.reject(err);
                        activeRequests--;
                        processQueue();
                    }
                });
        }
    }
</script>
